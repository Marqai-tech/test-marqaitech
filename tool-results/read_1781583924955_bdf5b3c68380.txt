    81→const SETTINGS_KEY = 'nexus_hrms_settings';
    82→
    83→function loadSettings(): Record<string, string | number | boolean> {
    84→  if (typeof window === 'undefined') return {};
    85→  try {
    86→    const stored = localStorage.getItem(SETTINGS_KEY);
    87→    if (stored) return JSON.parse(stored);
    88→  } catch { /* ignore */ }
    89→  return {};
    90→}
    91→
    92→function saveSettings(settings: Record<string, string | number | boolean>) {
    93→  if (typeof window === 'undefined') return;
    94→  try {
    95→    localStorage.setItem(SETTINGS_KEY, JSON.stringify(settings));
    96→  } catch { /* ignore */ }
    97→}

   102→function getAuthHeaders() {
   103→  const token = typeof window !== 'undefined' ? localStorage.getItem('tb_token') : null;
   104→  return { 'Content-Type': 'application/json', ...(token ? { Authorization: `Bearer ${token}` } : {}) };
   105→}

   525→                      <input

   545→        {/* Save/Cancel */}
   546→        {editing && (

   549→              onClick={onReset}

   551→            >
   552→              Cancel
   553→            </button>

   555→              onClick={onSave}
   556→              disabled={saving}
   557→              className="px-6 py-2.5 rounded-lg bg-blue-500 text-white text-sm font-medium hover:bg-blue-600 disabled:opacity-50 shadow-sm shadow-blue-500/25 flex items-center gap-2 transition-colors"
   558→            >
   559→              <FiSave className="w-4 h-4" />
   560→              {saving ? 'Saving...' : 'Save Changes'}
   561→            </button>
   562→          </div>
   563→        )}
   564→      </div>
   565→    </div>
   566→  );

   604→function SettingsContent() {
   605→  const searchParams = useSearchParams();
   606→  const router = useRouter();
   607→  const { user } = useAuthStore();
   608→  const isAdmin = ['super_admin', 'tenant_admin', 'hr_admin'].includes(user?.role || '');
   609→
   610→  // ALL hooks must be called before any conditional returns (Rules of Hooks)
   611→
   612→  // Read tab from URL query parameter
   613→  const tabFromUrl = searchParams.get('tab') as TabKey | null;
   614→  const navItems = allNavItems;
   615→  const validTabs: TabKey[] = ['general', 'payroll', 'leave', 'attendance', 'employee', 'recruitment', 'performance', 'company', 'roles', 'notifications'];
   616→  const defaultTab = (tabFromUrl && validTabs.includes(tabFromUrl)) ? tabFromUrl : 'general';
   617→
   618→  const [activeTab, setActiveTab] = useState<TabKey>(defaultTab);
   619→  const [saving, setSaving] = useState(false);
   620→
   621→  // Sync activeTab with URL search params ONLY when URL actually changes
   622→  useEffect(() => {
   623→    const tabFromUrlUpdate = searchParams.get('tab') as TabKey | null;
   624→    if (tabFromUrlUpdate && validTabs.includes(tabFromUrlUpdate)) {
   625→      setActiveTab(tabFromUrlUpdate);
   626→    }
   627→  }, [searchParams]);
   628→
   629→  // Redirect non-admins to My Profile page
   630→  useEffect(() => {
   631→    if (!isAdmin) {
   632→      router.replace('/my-profile');
   633→    }
   634→  }, [isAdmin, router]);
   635→
   636→  const [tenant, setTenant] = useState<TenantInfo | null>(null);
   637→  const [mobileSidebarOpen, setMobileSidebarOpen] = useState(false);
   638→
   639→  /* General settings state */
   640→  const [editingSection, setEditingSection] = useState<string | null>(null);
   641→  const [generalForm, setGeneralForm] = useState({
   642→    companyName: '',
   643→    timezone: 'UTC',
   644→    dateFormat: 'MM/DD/YYYY',
   645→    currency: 'INR',
   646→  });
   647→
   648→  /* Module settings from localStorage */
   649→  const [settings, setSettings] = useState<Record<string, string | number | boolean>>({});
   650→  const [editDrafts, setEditDrafts] = useState<Record<string, Record<string, string | number | boolean>>>({});
   651→
   652→  /* Notification preferences */
   653→  const [notifPrefs, setNotifPrefs] = useState<NotifPref>(() => {
   654→    if (typeof window === 'undefined') return { email: true, leave: true, payroll: true, recruitment: true, workflow: true, system: true };
   655→    const stored = localStorage.getItem('nexus_notif_prefs');
   656→    if (stored) {
   657→      try { return JSON.parse(stored) as NotifPref; } catch { /* ignore */ }
   658→    }
   659→    return { email: true, leave: true, payroll: true, recruitment: true, workflow: true, system: true };
   660→  });
   661→
   662→  /* Load settings from localStorage on mount */
   663→  useEffect(() => {
   664→    const stored = loadSettings();
   665→    setSettings(stored);
   666→  }, []);
   667→
   668→  /* Fetch tenant info */
   669→  const fetchTenant = useCallback(async () => {
   670→    try {
   671→      const res = await fetch('/api/tenants?limit=1', { headers: getAuthHeaders() });
   672→      if (res.ok) {
   673→        const data = await res.json();
   674→        const t = data.tenants?.[0] || null;
   675→        setTenant(t);
   676→        if (t) {
   677→          setGeneralForm({
   678→            companyName: t.name || '',
   679→            timezone: t.timezone || 'UTC',
   680→            dateFormat: 'MM/DD/YYYY',
   681→            currency: t.currency || 'INR',
   682→          });
   683→        }
   684→      }
   685→    } catch {
   686→      /* ignore - tenant API might not be available */
   687→    }
   688→  }, []);
   689→
   690→  useEffect(() => { queueMicrotask(() => fetchTenant()); }, [fetchTenant]);
   691→
   692→  /* Get default value for a field */
   693→  const getDefaultValue = useCallback((key: string): string | number | boolean => {
   694→    const allSections = [...payrollSections, ...leaveSections, ...attendanceSections, ...employeeSections, ...recruitmentSections, ...performanceSections];
   695→    for (const section of allSections) {
   696→      const field = section.fields.find((f) => f.key === key);
   697→      if (field) return field.defaultValue;
   698→    }
   699→    return '';
   700→  }, []);
   701→
   702→  /* Get current setting value (with default fallback) */
   703→  const getSetting = useCallback((key: string): string | number | boolean => {
   704→    if (key in settings) return settings[key];
   705→    return getDefaultValue(key);
   706→  }, [settings, getDefaultValue]);
   707→
   708→  /* Start editing a section */
   709→  const handleStartEdit = useCallback((sectionKey: string) => {
   710→    // Build a draft from current settings
   711→    const allSections = [...payrollSections, ...leaveSections, ...attendanceSections, ...employeeSections, ...recruitmentSections, ...performanceSections];
   712→    const section = allSections.find((s) => s.title === sectionKey);
   713→    if (!section) return;
   714→    const draft: Record<string, string | number | boolean> = {};
   715→    for (const field of section.fields) {
   716→      draft[field.key] = getSetting(field.key);
   717→    }
   718→    setEditDrafts((prev) => ({ ...prev, [sectionKey]: draft }));
   719→    setEditingSection(sectionKey);
   720→  }, [getSetting]);
   721→
   722→  /* Update a draft field */
   723→  const handleDraftChange = useCallback((sectionKey: string, key: string, value: string | number | boolean) => {
   724→    setEditDrafts((prev) => ({
   725→      ...prev,
   726→      [sectionKey]: { ...(prev[sectionKey] || {}), [key]: value },
   727→    }));
   728→  }, []);
   729→
   730→  /* Cancel editing */
   731→  const handleCancelEdit = useCallback(() => {
   732→    setEditingSection(null);
   733→    setEditDrafts((prev) => {
   734→      const next = { ...prev };
   735→      if (editingSection) delete next[editingSection];
   736→      return next;
   737→    });
   738→  }, [editingSection]);
   739→
   740→  /* Save section settings */
   741→  const handleSaveSection = useCallback((sectionKey: string) => {
   742→    const draft = editDrafts[sectionKey];
   743→    if (!draft) return;
   744→    const newSettings = { ...settings, ...draft };
   745→    setSettings(newSettings);
   746→    saveSettings(newSettings);
   747→    setEditingSection(null);
   748→    setEditDrafts((prev) => {
   749→      const next = { ...prev };
   750→      delete next[sectionKey];
   751→      return next;
   752→    });
   753→    toast.success('Settings saved successfully');
   754→  }, [editDrafts, settings]);
   755→
   756→  /* Save general settings */
   757→  const handleSaveGeneral = async () => {
   758→    try {
   759→      setSaving(true);
   760→      if (tenant?.id) {
   761→        const res = await fetch(`/api/tenants/${tenant.id}`, {
   762→          method: 'PATCH',
   763→          headers: getAuthHeaders(),
   764→          body: JSON.stringify({
   765→            name: generalForm.companyName,
   766→            timezone: generalForm.timezone,
   767→            currency: generalForm.currency,
   768→          }),
   769→        });
   770→        if (!res.ok) {
   771→          const d = await res.json();
   772→          throw new Error(d.error || 'Failed');
   773→        }
   774→      }
   775→      toast.success('Settings saved successfully');
   776→      setEditingSection(null);
   777→      fetchTenant();
   778→    } catch (err: unknown) {
   779→      toast.error(err instanceof Error ? err.message : 'Failed to save settings');
   780→    } finally {
   781→      setSaving(false);
   782→    }
   783→  };
   784→
   785→  /* Save notification preferences */
   786→  const handleSaveNotifPrefs = () => {
   787→    try {
   788→      localStorage.setItem('nexus_notif_prefs', JSON.stringify(notifPrefs));
   789→      toast.success('Notification preferences saved');
   790→    } catch {
   791→      toast.error('Failed to save preferences');
   792→    }
   793→  };
   794→
   795→  /* No longer needed — navItems are filtered by role, so employees never reach admin tabs */
   796→
   797→  // Early return for non-admins (after all hooks are called to satisfy Rules of Hooks)
   798→  if (!isAdmin) {
   799→    return (
   800→      <div className="flex items-center justify-center py-20">
   801→        <div className="text-center">
   802→          <div className="animate-spin w-8 h-8 border-2 border-blue-500 border-t-transparent rounded-full mx-auto mb-3" />
   803→          <p className="text-sm text-nexus-text-muted">Redirecting to My Profile...</p>
   804→        </div>
   805→      </div>
   806→    );
   807→  }
   808→
   809→  /* Build section cards for module settings tabs */
   810→  const renderModuleSections = (sections: SettingSection[]) => (
   811→    <div className="space-y-4">
   812→      {sections.map((section) => {
   813→        const isEditing = editingSection === section.title;
   814→        const draft = isEditing ? editDrafts[section.title] || {} : {};
   815→        const currentSettings: Record<string, string | number | boolean> = {};
   816→        for (const field of section.fields) {
   817→          currentSettings[field.key] = isEditing ? (draft[field.key] ?? getSetting(field.key)) : getSetting(field.key);
   818→        }
   819→        return (
   820→          <SettingsSectionCard
   821→            key={section.title}
   822→            section={section}
   823→            settings={currentSettings}
   824→            editing={isEditing}
   825→            onToggleEdit={() => {
   826→              if (isEditing) {
   827→                handleCancelEdit();
   828→              } else {
   829→                handleStartEdit(section.title);
   830→              }
   831→            }}
   832→            onChange={(key, value) => handleDraftChange(section.title, key, value)}
   833→            onSave={() => handleSaveSection(section.title)}
   834→            onReset={handleCancelEdit}
   835→            saving={saving}
   836→          />
   837→        );
   838→      })}
   839→    </div>
   840→  );
   841→
   842→  return (
   843→    <div className="space-y-6">
   844→      {/* Header */}
   845→      <div className="flex items-center justify-between">
   846→        <div>
   847→          <h1 className="text-2xl font-bold text-nexus-text-primary flex items-center gap-2">
   848→            <FiSettings className="w-6 h-6 text-slate-500" />
   849→            Settings
   850→          </h1>
   851→          <p className="text-nexus-text-secondary mt-1">Manage system configuration and preferences</p>
   852→        </div>
   853→        {/* Mobile tab selector */}
   854→        <button
   855→          onClick={() => setMobileSidebarOpen(!mobileSidebarOpen)}
   856→          className="md:hidden inline-flex items-center gap-2 px-3 py-2 rounded-lg border border-nexus-border text-sm font-medium text-nexus-text-secondary hover:bg-slate-50 transition-colors"
   857→        >
   858→          <FiMenu className="w-4 h-4" />
   859→          {navItems.find((n) => n.key === activeTab)?.label}
   860→        </button>
   861→      </div>
   862→
   863→      {/* Module Tips & Workflow — only shown for admins */}
   864→      {isAdmin && (
   865→      <div className="grid grid-cols-1 lg:grid-cols-2 gap-4">
   866→        <ModuleTips
   867→          moduleKey="settings"
   868→          title="Settings Tips"
   869→          tips={[
   870→            { title: 'Configure Modules First', description: 'Set up payroll, leave, and attendance rules before onboarding employees to ensure smooth operations.' },
   871→            { title: 'Role-Based Access', description: 'Use the Roles & Permissions tab to control who can view and modify different settings sections.' },
   872→            { title: 'Test Changes', description: 'After updating settings, verify the changes by checking the relevant module to ensure they take effect.' },
   873→            { title: 'Backup Settings', description: 'Settings are stored locally — consider documenting critical configurations for recovery purposes.' },
   874→            { title: 'Review Periodically', description: 'Revisit settings quarterly to align policies with evolving organizational needs and compliance requirements.' },
