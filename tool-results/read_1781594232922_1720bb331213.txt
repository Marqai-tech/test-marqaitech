   262→export function HelpTraining() {
   263→  const [searchQuery, setSearchQuery] = useState('');
   264→  const [selectedCategory, setSelectedCategory] = useState('all');
   265→  const [expandedModule, setExpandedModule] = useState<string | null>(null);
   266→  const [activeTab, setActiveTab] = useState('training');
   267→  const [completedTasks, setCompletedTasks] = useState<Record<string, boolean>>({});
   268→
   269→  const filteredModules = TRAINING_MODULES.filter(m => {
   270→    const matchesSearch = m.title.toLowerCase().includes(searchQuery.toLowerCase()) ||
   271→      m.description.toLowerCase().includes(searchQuery.toLowerCase());
   272→    const matchesCategory = selectedCategory === 'all' || m.category === selectedCategory;
   273→    return matchesSearch && matchesCategory;
   274→  });
   275→
   276→  const toggleTask = (moduleId: string, taskIndex: number) => {
   277→    const key = `${moduleId}-${taskIndex}`;
   278→    setCompletedTasks(prev => ({ ...prev, [key]: !prev[key] }));
   279→  };
   280→
   281→  const getTaskStatus = (moduleId: string, taskIndex: number) => {
   282→    return completedTasks[`${moduleId}-${taskIndex}`] || false;
   283→  };
   284→
   285→  const getModuleProgress = (module: TrainingModule) => {
   286→    const completed = module.tasks.filter((_, i) => getTaskStatus(module.id, i)).length;
   287→    return Math.round((completed / module.tasks.length) * 100);
   288→  };
   289→
   290→  return (
   291→    <div className="space-y-6">
   292→      {/* Header */}
   293→      <div className="flex flex-col sm:flex-row justify-between gap-4">
   294→        <div>
   295→          <h2 className="text-2xl font-bold flex items-center gap-2">
   296→            <BookOpen className="h-6 w-6 text-emerald-600" />
   297→            Help & Training Center

   299→          <p className="text-muted-foreground mt-1">Master every module with training videos, SOPs, and technical documentation</p>
   300→        </div>
   301→        <div className="flex items-center gap-2">
   302→          <Badge variant="secondary" className="bg-emerald-100 text-emerald-700">
   303→            {Object.keys(completedTasks).filter(k => completedTasks[k]).length} tasks completed
   304→          </Badge>
   305→        </div>
   306→      </div>
   307→
   308→      {/* Search and Filter */}
   309→      <div className="flex flex-col sm:flex-row gap-3">
   310→        <div className="relative flex-1">
   311→          <Search className="absolute left-3 top-1/2 -translate-y-1/2 h-4 w-4 text-muted-foreground" />
   312→          <Input
   313→            placeholder="Search modules, features, or workflows..."

   315→            value={searchQuery}
   316→            onChange={(e) => setSearchQuery(e.target.value)}
   317→          />
   318→        </div>
   319→        <div className="flex gap-2 flex-wrap">
   320→          {CATEGORIES.map(cat => (
   321→            <Button

   324→              size="sm"
   325→              onClick={() => setSelectedCategory(cat.key)}
   326→              className={selectedCategory === cat.key ? 'bg-emerald-600 hover:bg-emerald-700' : ''}

   328→              {cat.label}

   331→        </div>
   332→      </div>
   333→
   334→      {/* Tabs */}

   348→        {/* Training Videos Tab */}
   349→        <TabsContent value="training" className="space-y-4 mt-4">
   350→          <div className="grid grid-cols-1 md:grid-cols-2 lg:grid-cols-3 gap-4">
   351→            {filteredModules.map(module => (
   352→              <Card
   353→                key={module.id}
   354→                className={`cursor-pointer transition-all hover:shadow-md ${expandedModule === module.id ? 'ring-2 ring-emerald-500' : ''}`}
