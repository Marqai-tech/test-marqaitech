    60→    60→const formatDate = (dateStr: string | null | undefined) => {
    61→    61→  if (!dateStr) return '—'

    64→    64→
    65→    65→const formatStatus = (status: string) => {
    66→    66→  const map: Record<string, string> = {
    67→    67→    'active': 'Active', 'inactive': 'Inactive', 'onboarding': 'Onboarding',
    68→    68→    'exited': 'Exited', 'on_leave': 'On Leave', 'probation': 'Probation',
    69→    69→  }
    70→    70→  return map[status] || status.replace(/_/g, ' ').replace(/\b\w/g, c => c.toUpperCase())
    71→    71→}

    73→    73→function OrgNode({ node, depth = 0 }: { node: { name: string; children?: any[] }; depth?: number }) {

    96→    96→}
