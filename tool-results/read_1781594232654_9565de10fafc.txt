   146→export default function SelfServiceModule() {
   147→  const { data: session } = useSession()
   148→  const { currentCompanyId } = useHRMSStore()
   149→  const { toast } = useToast()
   150→
   151→  const [activeTab, setActiveTab] = useState('profile')
   152→  const [loading, setLoading] = useState(true)
   153→  const [employee, setEmployee] = useState<EmployeeData | null>(null)
   154→  const [employees, setEmployees] = useState<EmployeeData[]>([])
   155→  const [leaveTypes, setLeaveTypes] = useState<LeaveType[]>([])
   156→  const [leaveApps, setLeaveApps] = useState<LeaveApplication[]>([])
   157→  const [attendanceRecords, setAttendanceRecords] = useState<AttendanceRecord[]>([])
   158→  const [attendanceStats, setAttendanceStats] = useState({ present: 0, absent: 0, late: 0, halfDay: 0, total: 0 })
   159→  const [payslipRuns, setPayslipRuns] = useState<PayrollRunData[]>([])
   160→  const [myPayslips, setMyPayslips] = useState<PayrollTransactionData[]>([])
   161→  const [documents, setDocuments] = useState<EmployeeDocument[]>([])
   162→  const [tickets, setTickets] = useState<any[]>([])
   163→
   164→  // Leave application dialog
   165→  const [leaveDialog, setLeaveDialog] = useState(false)
   166→  const [leaveForm, setLeaveForm] = useState({ leaveTypeId: '', startDate: '', endDate: '', reason: '', halfDay: false })
   167→  const [submittingLeave, setSubmittingLeave] = useState(false)
   168→
   169→  // Attendance dialog
   170→  const [attendanceDialog, setAttendanceDialog] = useState(false)
   171→  const [attendanceForm, setAttendanceForm] = useState({ date: new Date().toISOString().split('T')[0], punchIn: '09:00', punchOut: '18:00', status: 'present' })
   172→  const [submittingAttendance, setSubmittingAttendance] = useState(false)
   173→
   174→  // Request dialog
   175→  const [requestDialog, setRequestDialog] = useState(false)
   176→  const [requestForm, setRequestForm] = useState({ category: '', priority: '', subject: '', description: '' })
   177→  const [submittingRequest, setSubmittingRequest] = useState(false)
   178→
   179→  const companyId = currentCompanyId || (session?.user as any)?.companyId
   180→
   181→  // Find current employee from session
   182→  const currentEmployeeId = employees.find(e => e.email === session?.user?.email)?.id
   183→
   184→  const fetchData = useCallback(async () => {
   185→    if (!companyId) return
   186→    setLoading(true)
   187→    try {
   188→      const [empRes, leaveRes, attendRes, payrollRes] = await Promise.all([
   189→        apiGet<{ employees: EmployeeData[]; total: number }>('/api/employees', { companyId, limit: '100' }),
   190→        apiGet<{ leaveTypes: LeaveType[]; applications: LeaveApplication[] }>('/api/leave', { companyId, include: 'all' }),
   191→        apiGet<{ records: AttendanceRecord[]; stats: { present: number; absent: number; late: number; halfDay: number; total: number } }>('/api/attendance', { companyId }),
   192→        apiGet<{ runs: PayrollRunData[] }>('/api/payroll', { companyId, include: 'runs' }),
   193→      ])
   194→
   195→      setEmployees(empRes.employees)
   196→      setLeaveTypes(leaveRes.leaveTypes || [])
   197→      setLeaveApps(leaveRes.applications || [])
   198→      setAttendanceRecords(attendRes.records || [])
   199→      setAttendanceStats(attendRes.stats || { present: 0, absent: 0, late: 0, halfDay: 0, total: 0 })
   200→      setPayslipRuns(payrollRes.runs || [])
   201→
   202→      // Find current user's employee record
   203→      const currentEmp = empRes.employees.find((e: EmployeeData) => e.email === session?.user?.email)
   204→      if (currentEmp) {
   205→        setEmployee(currentEmp)
   206→
   207→        // Filter leave applications for this employee
   208→        // (will be done in render)
   209→
   210→        // Filter attendance for this employee
   211→        const myAttendance = (attendRes.records || []).filter((r: AttendanceRecord) => r.employeeId === currentEmp.id)
   212→        setAttendanceRecords(myAttendance)
   213→
   214→        // Compute attendance stats from my records
   215→        const myPresent = myAttendance.filter((r: AttendanceRecord) => r.status === 'present').length
   216→        const myAbsent = myAttendance.filter((r: AttendanceRecord) => r.status === 'absent').length
   217→        const myLate = myAttendance.filter((r: AttendanceRecord) => r.lateMark).length
   218→        const myHalfDay = myAttendance.filter((r: AttendanceRecord) => r.status === 'half_day').length
   219→        setAttendanceStats({ present: myPresent, absent: myAbsent, late: myLate, halfDay: myHalfDay, total: myAttendance.length })
   220→
   221→        // Filter payroll transactions for this employee
   222→        const myPayTxns: PayrollTransactionData[] = []
   223→        for (const run of payrollRes.runs || []) {
   224→          if (run.status === 'completed') {
   225→            const txns = (run.transactions || []).filter((t: PayrollTransactionData) => t.employeeId === currentEmp.id)
   226→            myPayTxns.push(...txns.map((t: PayrollTransactionData) => ({ ...t, runMonth: run.month, runYear: run.year, runProcessedDate: run.processedDate })))
   227→          }
   228→        }
   229→        setMyPayslips(myPayTxns)
   230→
   231→        // Fetch documents
   232→        try {
   233→          const docRes = await apiGet<{ employees: EmployeeData[] }>('/api/employees', { companyId, search: currentEmp.email, limit: '1' })
   234→          // Documents come from EmployeeDocument model - we'll use helpdesk tickets as a proxy for requests
   235→        } catch {}
   236→
   237→        // Fetch helpdesk tickets for this employee as "requests"
   238→        try {
   239→          const helpRes = await apiGet<{ tickets: any[] }>('/api/helpdesk', { companyId })
   240→          const myTickets = (helpRes.tickets || []).filter((t: any) => t.requesterId === currentEmp.id)
   241→          setTickets(myTickets)
   242→        } catch {}
   243→      }
   244→    } catch (error: any) {
   245→      console.error('Self-service fetch error:', error)
   246→      toast({ title: 'Error', description: 'Failed to load self-service data', variant: 'destructive' })
   247→    } finally {
   248→      setLoading(false)
   249→    }
   250→  }, [companyId, session?.user?.email, toast])
   251→
   252→  useEffect(() => { fetchData() }, [fetchData])
   253→
   254→  // ── Leave balance computation ──
   255→  const leaveBalance = leaveTypes.map(lt => {
   256→    const used = leaveApps
   257→      .filter(a => a.leaveTypeId === lt.id && (a.status === 'approved' || a.status === 'Approved') && a.employeeId === currentEmployeeId)
   258→      .reduce((sum, a) => sum + a.totalDays, 0)
   259→    return {
   260→      type: lt.name,
   261→      total: lt.maxPerYear,
   262→      used: Math.round(used),
   263→      balance: Math.max(0, lt.maxPerYear - Math.round(used)),
   264→      color: ['bg-emerald-500', 'bg-teal-500', 'bg-cyan-500', 'bg-purple-500', 'bg-amber-500'][leaveTypes.indexOf(lt) % 5],
   265→    }
   266→  })
   267→
   268→  // ── My leave applications ──
   269→  const myLeaves = leaveApps.filter(a => a.employeeId === currentEmployeeId)
   270→
   271→  // ── Submit leave ──
   272→  const handleSubmitLeave = async () => {
   273→    if (!leaveForm.leaveTypeId || !leaveForm.startDate || !leaveForm.endDate) {
   274→      toast({ title: 'Validation', description: 'Please fill all required fields', variant: 'destructive' })
   275→      return
   276→    }
   277→    setSubmittingLeave(true)
   278→    try {
   279→      const start = new Date(leaveForm.startDate)
   280→      const end = new Date(leaveForm.endDate)
   281→      const days = Math.max(1, Math.ceil((end.getTime() - start.getTime()) / (1000 * 60 * 60 * 24)) + 1)
   282→      await apiPost('/api/leave', {
   283→        companyId,
   284→        employeeId: currentEmployeeId,
   285→        leaveTypeId: leaveForm.leaveTypeId,
   286→        startDate: leaveForm.startDate,
   287→        endDate: leaveForm.endDate,
   288→        totalDays: days,
   289→        halfDay: leaveForm.halfDay,
   290→        reason: leaveForm.reason,
   291→      })
   292→      toast({ title: 'Success', description: 'Leave application submitted' })
   293→      setLeaveDialog(false)
   294→      setLeaveForm({ leaveTypeId: '', startDate: '', endDate: '', reason: '', halfDay: false })
   295→      fetchData()
   296→    } catch (error: any) {
   297→      toast({ title: 'Error', description: error.message || 'Failed to submit leave', variant: 'destructive' })
   298→    } finally {
   299→      setSubmittingLeave(false)
   300→    }
   301→  }
   302→
   303→  // ── Mark attendance ──
   304→  const handleMarkAttendance = async () => {
   305→    if (!attendanceForm.date) {
   306→      toast({ title: 'Validation', description: 'Please select a date', variant: 'destructive' })
   307→      return
   308→    }
   309→    setSubmittingAttendance(true)
   310→    try {
   311→      const punchInDT = new Date(`${attendanceForm.date}T${attendanceForm.punchIn}:00`)
   312→      const punchOutDT = new Date(`${attendanceForm.date}T${attendanceForm.punchOut}:00`)
   313→      const workedHours = Math.max(0, (punchOutDT.getTime() - punchInDT.getTime()) / (1000 * 60 * 60))
   314→      const isLate = attendanceForm.punchIn > '09:15'
   315→
   316→      await apiPost('/api/attendance', {
   317→        companyId,
   318→        employeeId: currentEmployeeId,
   319→        date: attendanceForm.date,
   320→        firstPunchIn: punchInDT.toISOString(),
   321→        lastPunchOut: punchOutDT.toISOString(),
   322→        workedHours: Math.round(workedHours * 100) / 100,
   323→        status: attendanceForm.status,
   324→        source: 'web',
   325→        lateMark: isLate,
   326→      })
   327→      toast({ title: 'Success', description: 'Attendance marked successfully' })
   328→      setAttendanceDialog(false)
   329→      fetchData()
   330→    } catch (error: any) {
   331→      toast({ title: 'Error', description: error.message || 'Failed to mark attendance', variant: 'destructive' })
   332→    } finally {
   333→      setSubmittingAttendance(false)
   334→    }
   335→  }
   336→
   337→  // ── Submit request (helpdesk ticket) ──
   338→  const handleSubmitRequest = async () => {
   339→    if (!requestForm.subject || !requestForm.category) {
   340→      toast({ title: 'Validation', description: 'Please fill category and subject', variant: 'destructive' })
   341→      return
   342→    }
   343→    setSubmittingRequest(true)
   344→    try {
   345→      await apiPost('/api/helpdesk', {
   346→        companyId,
   347→        requesterId: currentEmployeeId,
   348→        category: requestForm.category,
   349→        priority: requestForm.priority || 'medium',
   350→        subject: requestForm.subject,
   351→        description: requestForm.description,
   352→      })
   353→      toast({ title: 'Success', description: 'Request submitted successfully' })
   354→      setRequestDialog(false)
   355→      setRequestForm({ category: '', priority: '', subject: '', description: '' })
   356→      fetchData()
   357→    } catch (error: any) {
   358→      toast({ title: 'Error', description: error.message || 'Failed to submit request', variant: 'destructive' })
   359→    } finally {
   360→      setSubmittingRequest(false)
   361→    }
   362→  }
   363→
   364→  // ── Month name helper ──
   365→  const monthName = (m: number) => ['', 'Jan', 'Feb', 'Mar', 'Apr', 'May', 'Jun', 'Jul', 'Aug', 'Sep', 'Oct', 'Nov', 'Dec'][m] || ''
   366→
   367→  const initials = employee ? `${employee.firstName[0] || ''}${employee.lastName[0] || ''}`.toUpperCase() : '??'
   368→
   369→  if (loading) {
   370→    return (
   371→      <div className="flex items-center justify-center h-64">
   372→        <Loader2 className="w-8 h-8 animate-spin text-emerald-500" />
   373→        <span className="ml-3 text-slate-400">Loading self-service data...</span>
   374→      </div>
   375→    )
   376→  }
   377→
   378→  return (
   379→    <div className="space-y-6">
   380→      <div>
   381→        <h1 className="text-2xl font-bold text-white">Employee Self-Service</h1>
   382→        <p className="text-slate-400 mt-1">Access your information and raise requests</p>
   383→      </div>
   384→
   385→      <Tabs value={activeTab} onValueChange={setActiveTab}>
   386→        <TabsList className="bg-slate-900 border border-slate-800 p-1 h-auto flex-wrap">
   387→          <TabsTrigger value="profile" className="data-[state=active]:bg-emerald-600 data-[state=active]:text-white text-slate-400 text-xs px-3 py-1.5">My Profile</TabsTrigger>
   388→          <TabsTrigger value="leaves" className="data-[state=active]:bg-emerald-600 data-[state=active]:text-white text-slate-400 text-xs px-3 py-1.5">My Leaves</TabsTrigger>
   389→          <TabsTrigger value="payslips" className="data-[state=active]:bg-emerald-600 data-[state=active]:text-white text-slate-400 text-xs px-3 py-1.5">My Payslips</TabsTrigger>
   390→          <TabsTrigger value="attendance" className="data-[state=active]:bg-emerald-600 data-[state=active]:text-white text-slate-400 text-xs px-3 py-1.5">My Attendance</TabsTrigger>
   391→          <TabsTrigger value="documents" className="data-[state=active]:bg-emerald-600 data-[state=active]:text-white text-slate-400 text-xs px-3 py-1.5">My Documents</TabsTrigger>
   392→          <TabsTrigger value="requests" className="data-[state=active]:bg-emerald-600 data-[state=active]:text-white text-slate-400 text-xs px-3 py-1.5">Raise Request</TabsTrigger>
   393→        </TabsList>
   394→
   395→        {/* My Profile */}
   396→        <TabsContent value="profile" className="mt-4">
   397→          <Card className="bg-slate-900 border-slate-800">
   398→            <CardContent className="p-6">
   399→              {employee ? (
   400→                <div className="flex flex-col sm:flex-row items-start gap-6">
   401→                  <div className="w-20 h-20 rounded-full bg-gradient-to-br from-emerald-500 to-teal-600 flex items-center justify-center text-white text-2xl font-bold shrink-0">
   402→                    {initials}
   403→                  </div>
   404→                  <div className="flex-1">
   405→                    <h2 className="text-xl font-bold text-white">{employee.firstName} {employee.lastName}</h2>
   406→                    <p className="text-emerald-400 mt-1">{employee.designation?.name || 'N/A'}</p>
   407→                    <div className="grid grid-cols-2 sm:grid-cols-3 gap-4 mt-4">
   408→                      <div><p className="text-xs text-slate-500">Employee ID</p><p className="text-sm text-white">{employee.employeeCode}</p></div>
   409→                      <div><p className="text-xs text-slate-500">Department</p><p className="text-sm text-white">{employee.department?.name || 'N/A'}</p></div>
   410→                      <div><p className="text-xs text-slate-500">Location</p><p className="text-sm text-white">{employee.branch?.city || employee.city || 'N/A'}</p></div>
   411→                      <div><p className="text-xs text-slate-500">Date of Joining</p><p className="text-sm text-white">{employee.joiningDate ? fmtDate(employee.joiningDate) : 'N/A'}</p></div>
   412→                      <div><p className="text-xs text-slate-500">Email</p><p className="text-sm text-white">{employee.email}</p></div>
   413→                      <div><p className="text-xs text-slate-500">Phone</p><p className="text-sm text-white">{employee.phone || 'N/A'}</p></div>
   414→                      <div><p className="text-xs text-slate-500">Company</p><p className="text-sm text-white">{employee.company?.name || 'N/A'}</p></div>
   415→                      <div><p className="text-xs text-slate-500">Status</p><Badge className="bg-emerald-500/10 text-emerald-400 border-emerald-500/20">{fmtStatus(employee.status)}</Badge></div>
   416→                    </div>
   417→                  </div>
   418→                </div>
   419→              ) : (
   420→                <div className="text-center py-8 text-slate-400">
   421→                  <User className="w-12 h-12 mx-auto mb-3 text-slate-600" />
   422→                  <p>No employee record found for your account.</p>
   423→                  <p className="text-xs text-slate-500 mt-1">Please contact HR to link your employee profile.</p>
   424→                </div>
   425→              )}
   426→            </CardContent>
   427→          </Card>
   428→        </TabsContent>
   429→
   430→        {/* My Leaves */}
   431→        <TabsContent value="leaves" className="mt-4">
   432→          <div className="flex justify-end mb-4">
   433→            <Dialog open={leaveDialog} onOpenChange={setLeaveDialog}>

   435→                <Button className="bg-emerald-600 hover:bg-emerald-700 text-white gap-2">

   440→                <DialogHeader>
   441→                  <DialogTitle className="text-white">Apply for Leave</DialogTitle>
   442→                </DialogHeader>
   443→                <div className="space-y-4 py-2">

   445→                    <Label className="text-slate-300">Leave Type</Label>
   446→                    <Select value={leaveForm.leaveTypeId} onValueChange={v => setLeaveForm(p => ({ ...p, leaveTypeId: v }))}>
