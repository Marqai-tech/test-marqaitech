     1→// ============================================================================
     2→// Role-Based Tips & Workflows Configuration for HRMS Application
     3→// ============================================================================
     4→
     5→/**
     6→ * A single tip for a module, optionally scoped to specific roles.
     7→ * If `roles` is undefined, the tip applies to all roles.
     8→ */
     9→export interface ModuleTip {
    10→  title: string;
    11→  description: string;
    12→  roles?: string[];
    13→}
    14→
    15→/**
    16→ * A single step in a module workflow, optionally scoped to specific roles.
    17→ * If `roles` is undefined, the step applies to all roles.
    18→ * Steps with the same `step` number but different roles represent
    19→ * role-specific alternatives at that point in the workflow.
    20→ */
    21→export interface WorkflowStep {
    22→  step: number;
    23→  title: string;
    24→  description: string;
    25→  route?: string;
    26→  roles?: string[];
    27→}
    28→
    29→/**
    30→ * Configuration for a single module's tips and workflow.
    31→ */
    32→export interface RoleTipsConfig {
    33→  moduleKey: string;
    34→  tips: ModuleTip[];
    35→  workflowSteps: WorkflowStep[];
    36→  workflowTitle: string;
    37→  workflowSubtitle?: string;
    38→}
    39→
