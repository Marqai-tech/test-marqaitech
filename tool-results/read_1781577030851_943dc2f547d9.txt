    88→const SETTINGS_KEY = 'nexus_hrms_settings';
    89→
    90→function loadSettings(): Record<string, string | number | boolean> {
    91→  if (typeof window === 'undefined') return {};
    92→  try {
    93→    const stored = localStorage.getItem(SETTINGS_KEY);
    94→    if (stored) return JSON.parse(stored);
    95→  } catch { /* ignore */ }
    96→  return {};
    97→}
    98→
    99→function saveSettings(settings: Record<string, string | number | boolean>) {
   100→  if (typeof window === 'undefined') return;
   101→  try {
   102→    localStorage.setItem(SETTINGS_KEY, JSON.stringify(settings));
   103→  } catch { /* ignore */ }
   104→}

   109→function getAuthHeaders() {
   110→  const token = typeof window !== 'undefined' ? localStorage.getItem('tb_token') : null;
   111→  return { 'Content-Type': 'application/json', ...(token ? { Authorization: `Bearer ${token}` } : {}) };
   112→}

   559→              Cancel
   560→            </button>

   562→              onClick={onSave}
   563→              disabled={saving}

   565→            >
   566→              <FiSave className="w-4 h-4" />
   567→              {saving ? 'Saving...' : 'Save Changes'}
   568→            </button>
   569→          </div>
   570→        )}
   571→      </div>
   572→    </div>
   573→  );
   574→}

   617→function SettingsContent() {
   618→  const searchParams = useSearchParams();
   619→  const { user } = useAuthStore();
   620→  const isAdmin = ['super_admin', 'tenant_admin', 'hr_admin'].includes(user?.role || '');

   640→  const [tenant, setTenant] = useState<TenantInfo | null>(null);

   642→
   643→  /* Profile state */
   644→  const [profileData, setProfileData] = useState<{
   645→    name: string;
   646→    email: string;
   647→    phone: string;
   648→    emergencyContactName: string;
   649→    emergencyContactPhone: string;
   650→    address: string;
   651→    city: string;
   652→    state: string;
   653→    zipCode: string;
   654→    country: string;
   655→    dateOfBirth: string;

   663→  } | null>(null);

   679→  const [settings, setSettings] = useState<Record<string, string | number | boolean>>({});
   680→  const [editDrafts, setEditDrafts] = useState<Record<string, Record<string, string | number | boolean>>>({});
   681→
   682→  /* Notification preferences */
   683→  const [notifPrefs, setNotifPrefs] = useState<NotifPref>(() => {
   684→    if (typeof window === 'undefined') return { email: true, leave: true, payroll: true, recruitment: true, workflow: true, system: true };
   685→    const stored = localStorage.getItem('nexus_notif_prefs');
   686→    if (stored) {
   687→      try { return JSON.parse(stored) as NotifPref; } catch { /* ignore */ }
   688→    }

   718→  }, []);
   719→
   720→  useEffect(() => { queueMicrotask(() => fetchTenant()); }, [fetchTenant]);
   721→

   724→    try {
   725→      setProfileLoading(true);
   726→      setProfileError(null);

   740→          country: data.employee?.country || '',

   745→          designation: data.employee?.designation || '',
   746→          branch: data.employee?.branch || '',

   748→          dateOfJoining: data.employee?.dateOfJoining || '',

   751→        const errData = await res.json().catch(() => ({}));
   752→        setProfileError(errData.error || 'Failed to load profile. Please try again.');

   754→    } catch {
   755→      setProfileError('Network error. Please check your connection and try again.');
   756→    } finally {
   757→      setProfileLoading(false);
   758→    }
   759→  }, []);

   766→      setProfileSaving(true);
   767→      const res = await fetch('/api/users/profile', {
   768→        method: 'PATCH',
   769→        headers: getAuthHeaders(),

   772→          phone: profileData?.phone,

   774→          emergencyContactPhone: profileData?.emergencyContactPhone,
   775→          address: profileData?.address,
   776→          city: profileData?.city,
   777→          state: profileData?.state,
   778→          zipCode: profileData?.zipCode,
   779→          country: profileData?.country,
   780→          dateOfBirth: profileData?.dateOfBirth || null,
   781→          gender: profileData?.gender,
   782→          bloodGroup: profileData?.bloodGroup,
   783→        }),
   784→      });
   785→      if (res.ok) {
   786→        toast.success('Profile updated successfully');

   790→        const data = await res.json();
   791→        toast.error(data.error || 'Failed to update profile');
   792→      }
   793→    } catch {
   794→      toast.error('Failed to update profile');
   795→    } finally {
   796→      setProfileSaving(false);
   797→    }
   798→  };
   799→
   800→  /* Get default value for a field */
   801→  const getDefaultValue = useCallback((key: string): string | number | boolean => {
   802→    const allSections = [...payrollSections, ...leaveSections, ...attendanceSections, ...employeeSections, ...recruitmentSections, ...performanceSections];
   803→    for (const section of allSections) {
   804→      const field = section.fields.find((f) => f.key === key);
   805→      if (field) return field.defaultValue;
   806→    }
   807→    return '';
   808→  }, []);

   810→  /* Get current setting value (with default fallback) */
   811→  const getSetting = useCallback((key: string): string | number | boolean => {
   812→    if (key in settings) return settings[key];
   813→    return getDefaultValue(key);
   814→  }, [settings, getDefaultValue]);
   815→
   816→  /* Start editing a section */
   817→  const handleStartEdit = useCallback((sectionKey: string) => {
   818→    // Build a draft from current settings
   819→    const allSections = [...payrollSections, ...leaveSections, ...attendanceSections, ...employeeSections, ...recruitmentSections, ...performanceSections];
   820→    const section = allSections.find((s) => s.title === sectionKey);
   821→    if (!section) return;
   822→    const draft: Record<string, string | number | boolean> = {};
   823→    for (const field of section.fields) {
   824→      draft[field.key] = getSetting(field.key);
   825→    }
   826→    setEditDrafts((prev) => ({ ...prev, [sectionKey]: draft }));
   827→    setEditingSection(sectionKey);
   828→  }, [getSetting]);
   829→
   830→  /* Update a draft field */
   831→  const handleDraftChange = useCallback((sectionKey: string, key: string, value: string | number | boolean) => {
   832→    setEditDrafts((prev) => ({
   833→      ...prev,
   834→      [sectionKey]: { ...(prev[sectionKey] || {}), [key]: value },
   835→    }));
   836→  }, []);
   837→
   838→  /* Cancel editing */
   839→  const handleCancelEdit = useCallback(() => {
   840→    setEditingSection(null);
   841→    setEditDrafts((prev) => {
   842→      const next = { ...prev };
   843→      if (editingSection) delete next[editingSection];
   844→      return next;
   845→    });
   846→  }, [editingSection]);
   847→
   848→  /* Save section settings */
   849→  const handleSaveSection = useCallback((sectionKey: string) => {
   850→    const draft = editDrafts[sectionKey];
   851→    if (!draft) return;
   852→    const newSettings = { ...settings, ...draft };
   853→    setSettings(newSettings);
   854→    saveSettings(newSettings);
   855→    setEditingSection(null);
   856→    setEditDrafts((prev) => {
   857→      const next = { ...prev };
   858→      delete next[sectionKey];
   859→      return next;
   860→    });
   861→    toast.success('Settings saved successfully');
   862→  }, [editDrafts, settings]);
   863→
   864→  /* Save general settings */

   867→      setSaving(true);
   868→      if (tenant?.id) {
   869→        const res = await fetch(`/api/tenants/${tenant.id}`, {
   870→          method: 'PATCH',
   871→          headers: getAuthHeaders(),
   872→          body: JSON.stringify({
   873→            name: generalForm.companyName,
   874→            timezone: generalForm.timezone,
   875→            currency: generalForm.currency,
   876→          }),
   877→        });
   878→        if (!res.ok) {
   879→          const d = await res.json();
   880→          throw new Error(d.error || 'Failed');
   881→        }
   882→      }
   883→      toast.success('Settings saved successfully');
