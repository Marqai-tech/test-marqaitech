    88→const SETTINGS_KEY = 'nexus_hrms_settings';
    89→
    90→function loadSettings(): Record<string, string | number | boolean> {
    91→  if (typeof window === 'undefined') return {};
    92→  try {
    93→    const stored = localStorage.getItem(SETTINGS_KEY);
    94→    if (stored) return JSON.parse(stored);
    95→  } catch { /* ignore */ }
    96→  return {};
    97→}
    98→
    99→function saveSettings(settings: Record<string, string | number | boolean>) {
   100→  if (typeof window === 'undefined') return;
   101→  try {
   102→    localStorage.setItem(SETTINGS_KEY, JSON.stringify(settings));
   103→  } catch { /* ignore */ }
   104→}

   109→function getAuthHeaders() {
   110→  const token = typeof window !== 'undefined' ? localStorage.getItem('tb_token') : null;
   111→  return { 'Content-Type': 'application/json', ...(token ? { Authorization: `Bearer ${token}` } : {}) };
   112→}

   526→                        min={field.min}
   527→                        max={field.max}

   532→                      <input
   533→                        type="text"
   534→                        value={String(settings[field.key] ?? field.defaultValue)}

   547→              )}
   548→            </div>
   549→          ))}
   550→        </div>
   551→
   552→        {/* Save/Cancel */}
   553→        {editing && (
   554→          <div className="flex items-center justify-end gap-3 mt-6 pt-4 border-t border-nexus-border">
   555→            <button
   556→              onClick={onReset}
   557→              className="px-4 py-2.5 rounded-lg border border-nexus-border text-sm font-medium text-nexus-text-secondary hover:bg-slate-50 transition-colors"
   558→            >
   559→              Cancel
   560→            </button>
   561→            <button
   562→              onClick={onSave}
   563→              disabled={saving}
   564→              className="px-6 py-2.5 rounded-lg bg-blue-500 text-white text-sm font-medium hover:bg-blue-600 disabled:opacity-50 shadow-sm shadow-blue-500/25 flex items-center gap-2 transition-colors"
   565→            >
   566→              <FiSave className="w-4 h-4" />
   567→              {saving ? 'Saving...' : 'Save Changes'}
   568→            </button>
   569→          </div>
   570→        )}
   571→      </div>
   572→    </div>
   573→  );
   574→}

   611→function SettingsContent() {
   612→  const searchParams = useSearchParams();
   613→  const router = useRouter();
   614→  const { user } = useAuthStore();
   615→  const isAdmin = ['super_admin', 'tenant_admin', 'hr_admin'].includes(user?.role || '');

   650→  }, [searchParams, allowedTabKeys]);
   651→  const [tenant, setTenant] = useState<TenantInfo | null>(null);
   652→  const [mobileSidebarOpen, setMobileSidebarOpen] = useState(false);
   653→
   654→  /* Profile state */
   655→  const [profileData, setProfileData] = useState<{

   657→    email: string;
   658→    phone: string;

   674→  } | null>(null);

   676→  const [profileSaving, setProfileSaving] = useState(false);
   677→  const [editingProfile, setEditingProfile] = useState(false);
   678→  const [profileError, setProfileError] = useState<string | null>(null);
   679→
   680→  /* General settings state */
   681→  const [editingSection, setEditingSection] = useState<string | null>(null);

   684→    timezone: 'UTC',
   685→    dateFormat: 'MM/DD/YYYY',
   686→    currency: 'INR',

   689→  /* Module settings from localStorage */
   690→  const [settings, setSettings] = useState<Record<string, string | number | boolean>>({});
   691→  const [editDrafts, setEditDrafts] = useState<Record<string, Record<string, string | number | boolean>>>({});
   692→
   693→  /* Notification preferences */

   727→      /* ignore - tenant API might not be available */
   728→    }

   768→      setProfileLoading(false);

   787→          city: profileData?.city,
   788→          state: profileData?.state,
   789→          zipCode: profileData?.zipCode,
   790→          country: profileData?.country,
   791→          dateOfBirth: profileData?.dateOfBirth || null,

   793→          bloodGroup: profileData?.bloodGroup,

   805→      toast.error('Failed to update profile');
   806→    } finally {
   807→      setProfileSaving(false);
   808→    }

   811→  /* Get default value for a field */
   812→  const getDefaultValue = useCallback((key: string): string | number | boolean => {
   813→    const allSections = [...payrollSections, ...leaveSections, ...attendanceSections, ...employeeSections, ...recruitmentSections, ...performanceSections];

   831→    const section = allSections.find((s) => s.title === sectionKey);
   832→    if (!section) return;
   833→    const draft: Record<string, string | number | boolean> = {};
   834→    for (const field of section.fields) {
   835→      draft[field.key] = getSetting(field.key);
   836→    }
   837→    setEditDrafts((prev) => ({ ...prev, [sectionKey]: draft }));
   838→    setEditingSection(sectionKey);
   839→  }, [getSetting]);
   840→

   845→      [sectionKey]: { ...(prev[sectionKey] || {}), [key]: value },

   854→      if (editingSection) delete next[editingSection];
   855→      return next;
   856→    });

   865→    saveSettings(newSettings);
   866→    setEditingSection(null);
   867→    setEditDrafts((prev) => {
   868→      const next = { ...prev };
   869→      delete next[sectionKey];
   870→      return next;
   871→    });

   873→  }, [editDrafts, settings]);

   878→      setSaving(true);

   880→        const res = await fetch(`/api/tenants/${tenant.id}`, {

   882→          headers: getAuthHeaders(),
   883→          body: JSON.stringify({
   884→            name: generalForm.companyName,
   885→            timezone: generalForm.timezone,
