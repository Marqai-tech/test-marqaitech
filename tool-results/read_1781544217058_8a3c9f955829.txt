    23→function getAuthHeaders() {
    24→  const token = typeof window !== 'undefined' ? localStorage.getItem('tb_token') : null;
    25→  return { 'Content-Type': 'application/json', ...(token ? { Authorization: `Bearer ${token}` } : {}) };
    26→}

    28→// ─── Employee Policies Tab Component ───
    29→function EmployeePoliciesTab({ employeeId }: { employeeId: string }) {
    30→  const { user } = useAuthStore();
    31→  const isAdmin = user?.role === 'super_admin' || user?.role === 'tenant_admin' || user?.role === 'hr_admin';
    32→  const [policies, setPolicies] = useState<{ leave?: Record<string, unknown>; attendance?: Record<string, unknown>; travel?: Record<string, unknown>; salaryStructure?: Record<string, unknown>; leavePolicyId?: string | null; attendancePolicyId?: string | null; travelPolicyId?: string | null; salaryStructureId?: string | null } | null>(null);
    33→  const [leavePolicies, setLeavePolicies] = useState<Array<{ id: string; title: string; version: string; effectiveDate?: string | null }>>([]);
    34→  const [attendancePolicies, setAttendancePolicies] = useState<Array<{ id: string; title: string; version: string; effectiveDate?: string | null }>>([]);
    35→  const [travelPolicies, setTravelPolicies] = useState<Array<{ id: string; title: string; version: string; effectiveDate?: string | null }>>([]);
    36→  const [salaryStructures, setSalaryStructures] = useState<Array<{ id: string; name: string }>>([]);
    37→  const [saving, setSaving] = useState<string | null>(null);
    38→
    39→  const fetchPolicies = useCallback(async () => {
    40→    try {
    41→      const [polRes, lpRes, apRes, tpRes, ssRes] = await Promise.allSettled([
    42→        fetch(`/api/employees/${employeeId}/policies`, { headers: getAuthHeaders() }),
    43→        fetch('/api/policies?category=leave', { headers: getAuthHeaders() }),
    44→        fetch('/api/policies?category=attendance', { headers: getAuthHeaders() }),
    45→        fetch('/api/policies?category=travel', { headers: getAuthHeaders() }),
    46→        fetch('/api/salary-structures', { headers: getAuthHeaders() }),
    47→      ]);
    48→      if (polRes.status === 'fulfilled' && polRes.value.ok) { const d = await polRes.value.json(); setPolicies(d.data); }
    49→      if (lpRes.status === 'fulfilled' && lpRes.value.ok) { const d = await lpRes.value.json(); setLeavePolicies((d.policies || d.data || [])); }
    50→      if (apRes.status === 'fulfilled' && apRes.value.ok) { const d = await apRes.value.json(); setAttendancePolicies((d.policies || d.data || [])); }
    51→      if (tpRes.status === 'fulfilled' && tpRes.value.ok) { const d = await tpRes.value.json(); setTravelPolicies((d.policies || d.data || [])); }
    52→      if (ssRes.status === 'fulfilled' && ssRes.value.ok) { const d = await ssRes.value.json(); setSalaryStructures((d.salaryStructures || d.data || [])); }
    53→    } catch { /* silently */ }
    54→  }, [employeeId]);
    55→
    56→  useEffect(() => { queueMicrotask(() => fetchPolicies()); }, [fetchPolicies]);
    57→
    58→  const handleSave = async (field: string, value: string | null) => {
    59→    setSaving(field);
    60→    try {
    61→      const res = await fetch(`/api/employees/${employeeId}/policies`, {
    62→        method: 'PUT', headers: getAuthHeaders(),
    63→        body: JSON.stringify({ [field]: value }),
    64→      });
    65→      if (res.ok) { toast.success('Updated'); fetchPolicies(); }
    66→      else { toast.error('Failed'); }
    67→    } catch { toast.error('Failed'); }
    68→    finally { setSaving(null); }
    69→  };
    70→
    71→  const policyDropdown = (label: string, field: string, items: Array<{ id: string; title: string; version: string; effectiveDate?: string | null }>, currentId: string | null | undefined, icon: React.ReactNode) => (
    72→    <div className="nexus-card p-4">
    73→      <h4 className="text-sm font-semibold text-nexus-text-primary mb-3 flex items-center gap-2">{icon}{label}</h4>
    74→      <select
    75→        value={currentId || ''}
    76→        onChange={e => handleSave(field, e.target.value || null)}
    77→        disabled={!isAdmin || saving === field}
    78→        className="w-full px-3 py-2 rounded-lg border border-nexus-border text-sm focus:outline-none focus:ring-2 focus:ring-blue-500/20 disabled:opacity-50"
    79→      >
    80→        <option value="">— Not Assigned —</option>
    81→        {items.map(p => <option key={p.id} value={p.id}>{p.title} (v{p.version})</option>)}
    82→      </select>
    83→      {currentId && items.find(p => p.id === currentId) && (
    84→        <div className="mt-3 p-3 rounded-lg bg-slate-50 border border-nexus-border text-xs text-nexus-text-secondary">
    85→          <p><span className="font-medium">Title:</span> {items.find(p => p.id === currentId)?.title}</p>
    86→          <p><span className="font-medium">Version:</span> {items.find(p => p.id === currentId)?.version}</p>
    87→          {items.find(p => p.id === currentId)?.effectiveDate && <p><span className="font-medium">Effective:</span> {new Date(items.find(p => p.id === currentId)!.effectiveDate!).toLocaleDateString()}</p>}
    88→        </div>
    89→      )}
    90→      {saving === field && <p className="text-xs text-blue-500 mt-2">Saving...</p>}
    91→    </div>
    92→  );
    93→
    94→  return (
    95→    <div className="space-y-4">
    96→      <h3 className="text-sm font-semibold text-nexus-text-primary flex items-center gap-2"><FiShield className="w-4 h-4 text-violet-500" />Policy & Structure Mappings</h3>
    97→      {!isAdmin && <div className="p-3 rounded-lg bg-amber-50 border border-amber-100 text-xs text-amber-700">Only admins can modify policy mappings.</div>}
    98→      <div className="grid grid-cols-1 sm:grid-cols-2 gap-4">
    99→        {policyDropdown('Leave Policy', 'leavePolicyId', leavePolicies, policies?.leavePolicyId, <FiCalendar className="w-4 h-4 text-blue-500" />)}
   100→        {policyDropdown('Attendance Policy', 'attendancePolicyId', attendancePolicies, policies?.attendancePolicyId, <FiClock className="w-4 h-4 text-emerald-500" />)}
   101→        {policyDropdown('Travel Policy', 'travelPolicyId', travelPolicies, policies?.travelPolicyId, <FiBriefcase className="w-4 h-4 text-amber-500" />)}
   102→        <div className="nexus-card p-4">
   103→          <h4 className="text-sm font-semibold text-nexus-text-primary mb-3 flex items-center gap-2"><FiDollarSign className="w-4 h-4 text-emerald-500" />Salary Structure</h4>
   104→          <select
   105→            value={policies?.salaryStructureId || ''}
   106→            onChange={e => handleSave('salaryStructureId', e.target.value || null)}
   107→            disabled={!isAdmin || saving === 'salaryStructureId'}
   108→            className="w-full px-3 py-2 rounded-lg border border-nexus-border text-sm focus:outline-none focus:ring-2 focus:ring-blue-500/20 disabled:opacity-50"
   109→          >
   110→            <option value="">— Not Assigned —</option>
   111→            {salaryStructures.map(s => <option key={s.id} value={s.id}>{s.name}</option>)}
   112→          </select>
   113→          {saving === 'salaryStructureId' && <p className="text-xs text-blue-500 mt-2">Saving...</p>}
   114→        </div>
   115→      </div>
   116→    </div>
   117→  );
   118→}

   120→function formatDate(d: string | null | undefined) {
   121→  if (!d) return '—';
   122→  return new Date(d).toLocaleDateString('en-US', { year: 'numeric', month: 'short', day: 'numeric' });
   123→}
   124→
   125→function formatDateTime(d: string | null | undefined) {
   126→  if (!d) return '—';
   127→  return new Date(d).toLocaleString('en-US', { year: 'numeric', month: 'short', day: 'numeric', hour: '2-digit', minute: '2-digit' });
   128→}

   130→function getStatusBadge(status: string) {
   131→  const map: Record<string, string> = {
   132→    active: 'nexus-badge nexus-badge-success',
   133→    on_leave: 'nexus-badge nexus-badge-warning',
   134→    terminated: 'nexus-badge nexus-badge-error',
   135→    resigned: 'bg-slate-100 text-slate-600 nexus-badge',
   136→  };
   137→  const labels: Record<string, string> = { active: 'Active', on_leave: 'On Leave', terminated: 'Terminated', resigned: 'Resigned' };
   138→  return { className: map[status] || 'nexus-badge nexus-badge-info', label: labels[status] || status };
   139→}

   189→const tabs = ['Overview', 'Personal', 'Employment', 'Documents', 'Leave', 'Attendance', 'Payroll', 'Health', 'Policies'] as const;
   190→type TabName = typeof tabs[number];

   192→export default function EmployeeDetailPage() {
   193→  const params = useParams();
   194→  const router = useRouter();
   195→  const id = params.id as string;
   196→
   197→  const [employee, setEmployee] = useState<EmployeeData | null>(null);
   198→  const [loading, setLoading] = useState(true);
   199→  const [activeTab, setActiveTab] = useState<TabName>('Overview');
   200→
   201→  const [attendance, setAttendance] = useState<Array<{ id: string; date: string; checkIn?: string | null; checkOut?: string | null; workHours?: number | null; overtime?: number | null; status: string }>>([]);
   202→  const [payrolls, setPayrolls] = useState<Array<{ id: string; month: number; year: number; grossSalary: number; totalDeductions: number; netSalary: number; status: string; paidDate?: string | null }>>([]);
   203→  const [leaveRequests, setLeaveRequests] = useState<Array<{ id: string; startDate: string; endDate: string; reason?: string | null; status: string; halfDay: boolean; leaveType: { name: string } }>>([]);
   204→  const [now] = useState(() => Date.now());
   205→
   206→  const fetchEmployee = useCallback(async () => {
   207→    try {
   208→      setLoading(true);
   209→      const res = await fetch(`/api/employees/${id}`, { headers: getAuthHeaders() });
   210→      if (!res.ok) throw new Error('Failed to fetch employee');
   211→      const data = await res.json();
   212→      setEmployee(data.employee);
   213→    } catch (err) {
   214→      console.error('Failed to load employee details:', err);
   215→      // Don't show error toast - the page shows a "not found" state that handles this gracefully
   216→      setEmployee(null);
   217→    } finally {
   218→      setLoading(false);
   219→    }
   220→  }, [id]);
   221→
   222→  const fetchRelatedData = useCallback(async () => {
   223→    if (!employee) return;
   224→    try {
   225→      const headers = getAuthHeaders();
   226→      const [attRes, payRes, leaveRes] = await Promise.allSettled([
   227→        fetch(`/api/attendance?employeeId=${employee.id}&limit=30`, { headers }),
   228→        fetch(`/api/payroll?employeeId=${employee.id}&limit=12`, { headers }),
   229→        fetch(`/api/leave?employeeId=${employee.id}&limit=20`, { headers }),
   230→      ]);
   231→      if (attRes.status === 'fulfilled' && attRes.value.ok) {
   232→        const d = await attRes.value.json();
   233→        setAttendance(d.attendance || []);
   234→      }
   235→      if (payRes.status === 'fulfilled' && payRes.value.ok) {
   236→        const d = await payRes.value.json();
   237→        setPayrolls(d.payrolls || []);
   238→      }
   239→      if (leaveRes.status === 'fulfilled' && leaveRes.value.ok) {
   240→        const d = await leaveRes.value.json();
   241→        setLeaveRequests(d.leaveRequests || []);
   242→      }
   243→    } catch (err) {
   244→      console.error(err);
   245→    }
   246→  }, [employee]);
   247→
   248→  useEffect(() => {
   249→    queueMicrotask(() => fetchEmployee());
   250→  }, [fetchEmployee]);
   251→
   252→  useEffect(() => {
   253→    if (employee) queueMicrotask(() => fetchRelatedData());
   254→  }, [employee, fetchRelatedData]);
   255→
   256→  if (loading) {
   257→    return (
   258→      <div className="space-y-6">
   259→        <div className="nexus-card p-6 animate-pulse">
   260→          <div className="flex items-center gap-4">
   261→            <div className="w-16 h-16 bg-slate-200 rounded-full" />
   262→            <div className="space-y-2"><div className="h-5 w-48 bg-slate-200 rounded" /><div className="h-3 w-32 bg-slate-200 rounded" /></div>
   263→          </div>
   264→        </div>
   265→        <div className="nexus-card p-6 animate-pulse"><div className="h-8 bg-slate-200 rounded w-full" /></div>
   266→      </div>
   267→    );
   268→  }
   269→
   270→  if (!employee) {
   271→    return (
   272→      <div className="text-center py-12">
   273→        <FiUser className="w-12 h-12 text-nexus-text-muted mx-auto mb-4" />
   274→        <p className="text-nexus-text-secondary font-medium">Employee not found</p>
   275→        <button onClick={() => router.push('/employees')} className="mt-4 text-blue-500 hover:text-blue-700 text-sm font-medium">Back to Employees</button>
   276→      </div>
   277→    );
   278→  }
   279→
   280→  const badge = getStatusBadge(employee.status);
   281→  const initials = `${employee.firstName[0]}${employee.lastName[0]}`;
   282→  const daysEmployed = Math.floor((now - new Date(employee.dateOfJoining).getTime()) / (1000 * 60 * 60 * 24));
   283→  const totalLeaveBalance = employee.leaveBalances?.reduce((sum, lb) => sum + lb.remaining, 0) || 0;
   284→  const attendanceRate = attendance.length > 0 ? Math.round((attendance.filter(a => ['present', 'late'].includes(a.status)).length / attendance.length) * 100) : 0;
   285→
   286→  return (

   290→        <FiArrowLeft className="w-4 h-4" />
   291→        Back to Employees
   292→      </button>
   293→
   294→      {/* Employee Header */}
   295→      <div className="nexus-card p-6">
   296→        <div className="flex flex-col sm:flex-row items-start sm:items-center gap-4">
   297→          {employee.avatar ? (
   298→            <img src={employee.avatar} alt="" className="w-16 h-16 rounded-full object-cover ring-4 ring-slate-100" />
   299→          ) : (
   300→            <div className="w-16 h-16 rounded-full bg-gradient-to-br from-blue-500 to-violet-600 flex items-center justify-center text-white text-xl font-bold ring-4 ring-slate-100 flex-shrink-0">
   301→              {initials}
   302→            </div>
   303→          )}
   304→          <div className="flex-1 min-w-0">

   311→              <span className="flex items-center gap-1"><FiMail className="w-3.5 h-3.5" />{employee.email}</span>
   312→              <span>{employee.department?.name} · {employee.designation?.title}</span>

   329→                    : 'border-transparent text-nexus-text-secondary hover:text-nexus-text-primary hover:border-slate-300'

   360→              {/* Quick Info */}
