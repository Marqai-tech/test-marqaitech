    98→/* ── Helpers ── */
    99→function getAuthHeaders() {
   100→  const token = typeof window !== 'undefined' ? localStorage.getItem('tb_token') : null;
   101→  return { 'Content-Type': 'application/json', ...(token ? { Authorization: `Bearer ${token}` } : {}) };
   102→}
   103→
   104→function getStatusBadge(status: string) {
   105→  const map: Record<string, string> = {
   106→    active: 'nexus-badge nexus-badge-success',
   107→    on_leave: 'nexus-badge nexus-badge-warning',
   108→    terminated: 'nexus-badge nexus-badge-error',
   109→    resigned: 'bg-slate-100 text-slate-600 nexus-badge',
   110→  };
   111→  const labels: Record<string, string> = { active: 'Active', on_leave: 'On Leave', terminated: 'Terminated', resigned: 'Resigned' };
   112→  return { className: map[status] || 'nexus-badge nexus-badge-info', label: labels[status] || status };
   113→}

   115→function formatDate(d: string | null | undefined) {
   116→  if (!d) return '—';
   117→  return new Date(d).toLocaleDateString('en-US', { year: 'numeric', month: 'short', day: 'numeric' });
   118→}
