name: Vercel Deploy (CLI, integration-independent)

# Deploys via the Vercel CLI using repo secrets — completely independent of
# the Vercel GitHub App integration. Use this when a project's Git connection
# is broken (e.g. the demo project hosted on an old Vercel account whose
# GitHub App authorization no longer covers the new GitHub account).
#
# SETUP (per project you want CI-deployed):
#   1. Vercel dashboard -> project -> Settings -> General:
#        - copy "Vercel ID"        (orgId)
#        - copy "Project ID"       (projectId)
#   2. Vercel dashboard -> Account Settings -> Tokens -> Create token
#      (scope: the account that hosts the project).
#   3. GitHub repo -> Settings -> Secrets and variables -> Actions -> add:
#        VERCEL_TOKEN                       (the token)
#        VERCEL_ORG_ID_DEMO / VERCEL_ORG_ID_LIVE        (orgId)
#        VERCEL_PROJECT_ID_DEMO / VERCEL_PROJECT_ID_LIVE (projectId)
#   Without the secrets this workflow skips cleanly (no red X).
#
# NOTE: add secrets to only ONE repo if both mirrors are watched by the same
# Vercel project, to avoid duplicate deployments.

on:
  push:
    branches: [main]
  workflow_dispatch:

jobs:
  deploy-demo:
    name: Deploy demo (nexus-hrms project)
    runs-on: ubuntu-latest
    steps:
      - uses: actions/checkout@v4
      - name: Skip — demo secrets not configured
        if: ${{ secrets.VERCEL_TOKEN == '' || secrets.VERCEL_ORG_ID_DEMO == '' || secrets.VERCEL_PROJECT_ID_DEMO == '' }}
        run: echo "VERCEL_TOKEN / VERCEL_ORG_ID_DEMO / VERCEL_PROJECT_ID_DEMO not set — skipping demo CLI deploy (git integration still active)."
      - name: Deploy demo to Vercel (production)
        if: ${{ secrets.VERCEL_TOKEN != '' && secrets.VERCEL_ORG_ID_DEMO != '' && secrets.VERCEL_PROJECT_ID_DEMO != '' }}
        env:
          VERCEL_TOKEN: ${{ secrets.VERCEL_TOKEN }}
          VERCEL_ORG_ID: ${{ secrets.VERCEL_ORG_ID_DEMO }}
          VERCEL_PROJECT_ID: ${{ secrets.VERCEL_PROJECT_ID_DEMO }}
        run: |
          mkdir -p .vercel
          printf '{"orgId":"%s","projectId":"%s"}' "$VERCEL_ORG_ID" "$VERCEL_PROJECT_ID" > .vercel/project.json
          npx --yes vercel@latest --prod --yes --token="$VERCEL_TOKEN"

  deploy-live:
    name: Deploy LIVE (3-boxes-hrms project)
    runs-on: ubuntu-latest
    steps:
      - uses: actions/checkout@v4
      - name: Skip — LIVE secrets not configured
        if: ${{ secrets.VERCEL_TOKEN_LIVE == '' || secrets.VERCEL_ORG_ID_LIVE == '' || secrets.VERCEL_PROJECT_ID_LIVE == '' }}
        run: echo "VERCEL_TOKEN_LIVE / VERCEL_ORG_ID_LIVE / VERCEL_PROJECT_ID_LIVE not set — skipping LIVE CLI deploy (git integration still active)."
      - name: Deploy LIVE to Vercel (production)
        if: ${{ secrets.VERCEL_TOKEN_LIVE != '' && secrets.VERCEL_ORG_ID_LIVE != '' && secrets.VERCEL_PROJECT_ID_LIVE != '' }}
        env:
          VERCEL_TOKEN: ${{ secrets.VERCEL_TOKEN_LIVE }}
          VERCEL_ORG_ID: ${{ secrets.VERCEL_ORG_ID_LIVE }}
          VERCEL_PROJECT_ID: ${{ secrets.VERCEL_PROJECT_ID_LIVE }}
        run: |
          mkdir -p .vercel
          printf '{"orgId":"%s","projectId":"%s"}' "$VERCEL_ORG_ID" "$VERCEL_PROJECT_ID" > .vercel/project.json
          npx --yes vercel@latest --prod --yes --token="$VERCEL_TOKEN_LIVE"
